The dev server runs on port 3116 and is killed at the end. `curl` output is stripped of `<script>` blocks, counts use `grep -o | wc -l` (the HTML is one line), and patterns avoid the apostrophe (React escapes it).

```
cd <repo root>
npm run dev -- --port 3116 > /tmp/dev-pd.log 2>&1 &  echo $! > /tmp/dev-pd.pid
until grep -q "Ready" /tmp/dev-pd.log; do sleep 1; done
P=http://localhost:3116
strip() { perl -0pe 's/<script.*?<\/script>//gs'; }
M=$(node -e 'console.log(encodeURIComponent(JSON.stringify({relationship:"niece-nephew",ageBand:"infants",ageYears:0,occasion:"naming",budgetBand:"under-2500",metal:"silver",symbols:"any",category:"any"})))')

# 1. Every route: title, lang, one main-content, one h1, skip link before the first h1, one new-tab hint
for path in "/" "/find" "/picks?m=$M" "/atelier" "/piece/TNZ-1056" "/piece/TNZ-1058" "/piece/NOPE-0000"; do
  H=$(curl -s "$P$path" | strip)
  echo "== $path"
  grep -o '<title>[^<]*</title>' <<<"$H"                                   # see list below
  grep -o '<html lang="en"' <<<"$H" | wc -l                                 # 1
  grep -o 'id="main-content"' <<<"$H" | wc -l                               # 1
  grep -o '<h1' <<<"$H" | wc -l                                             # 1
  grep -bo 'Skip to main content\|<h1' <<<"$H" | head -1                    # the Skip line comes first
  grep -o 'Opens in a new tab' <<<"$H" | wc -l                              # 1
done
# Titles expected:
#  /                       Haunted Gift Atelier — Tiny Tiaraa
```
